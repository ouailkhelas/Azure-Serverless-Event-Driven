# Useful CLI Commands for Testing and Monitoring

## Configuration Variables (Update these) 
RESOURCE_GROUP="rg-serverless-dev"
STORAGE_ACCOUNT="stguploaddev"
FUNCTION_APP="func-blob-processor-dev"
COSMOS_ACCOUNT="cosmos-serverless-dev"
COSMOS_DATABASE="ProcessedData"
COSMOS_CONTAINER="FileProcessingResults"
CONTAINER_NAME="uploads"

# ==========================================
# BLOB STORAGE OPERATIONS
# ==========================================

az storage blob list \
  --account-name "$STORAGE_ACCOUNT" \
  --container-name "$CONTAINER_NAME" \
  --output table

az storage blob upload \
  --account-name "$STORAGE_ACCOUNT" \
  --container-name "$CONTAINER_NAME" \
  --name "test-file.txt" \
  --file "./test-files/sample-document.txt" \
  --overwrite

az storage blob download \
  --account-name "$STORAGE_ACCOUNT" \
  --container-name "$CONTAINER_NAME" \
  --name "sample-document.txt" \
  --file "./downloaded-file.txt"

az storage blob delete \
  --account-name "$STORAGE_ACCOUNT" \
  --container-name "$CONTAINER_NAME" \
  --name "sample-document.txt"

az storage blob show \
  --account-name "$STORAGE_ACCOUNT" \
  --container-name "$CONTAINER_NAME" \
  --name "sample-document.txt"

# ==========================================
# FUNCTION APP MONITORING
# ==========================================

## View function app details
az functionapp show \
  --name "$FUNCTION_APP" \
  --resource-group "$RESOURCE_GROUP"

## List all functions in app
az functionapp function list \
  --name "$FUNCTION_APP" \
  --resource-group "$RESOURCE_GROUP"

## Get function details
az functionapp function show \
  --name "$FUNCTION_APP" \
  --resource-group "$RESOURCE_GROUP" \
  --function-name "BlobProcessor"

## View recent function logs (tail)
az functionapp log tail \
  --name "$FUNCTION_APP" \
  --resource-group "$RESOURCE_GROUP"

## View specific invocation logs
az functionapp log tail \
  --name "$FUNCTION_APP" \
  --resource-group "$RESOURCE_GROUP" \
  --max-lines 50

## Get function app connection strings
az functionapp config appsettings list \
  --name "$FUNCTION_APP" \
  --resource-group "$RESOURCE_GROUP"

# ==========================================
# EVENT GRID OPERATIONS
# ==========================================

## List Event Grid subscriptions
az eventgrid event-subscription list \
  --resource-name "$STORAGE_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --resource-type "Microsoft.Storage/storageAccounts"

## Get subscription details
az eventgrid event-subscription show \
  --name "blob-upload-to-function" \
  --resource-name "$STORAGE_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --resource-type "Microsoft.Storage/storageAccounts"

## Test Event Grid topic
az eventgrid topic event-subscription list \
  --topic-name "eventgrid-topic-dev" \
  --resource-group "$RESOURCE_GROUP"

# ==========================================
# COSMOS DB QUERIES
# ==========================================

## List Cosmos DB accounts
az cosmosdb show \
  --name "$COSMOS_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP"

## Get Cosmos DB connection string
az cosmosdb keys list \
  --name "$COSMOS_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --type connection-strings

## View database details
az cosmosdb sql database show \
  --account-name "$COSMOS_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --name "$COSMOS_DATABASE"

## View container details
az cosmosdb sql container show \
  --account-name "$COSMOS_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --database-name "$COSMOS_DATABASE" \
  --name "$COSMOS_CONTAINER"

# ==========================================
# RESOURCE MONITORING
# ==========================================

## Get resource group info
az group show \
  --name "$RESOURCE_GROUP"

## List all resources in group
az resource list \
  --resource-group "$RESOURCE_GROUP" \
  --output table

## View resource creation history
az monitor activity-log list \
  --resource-group "$RESOURCE_GROUP" \
  --query "[0:10]" \
  --output table

## Check resource costs
az consumption usage list \
  --output table

# ==========================================
# FUNCTION DEPLOYMENT
# ==========================================

## Deploy function from local code
func azure functionapp publish "$FUNCTION_APP" --python --build remote

## Deploy with build pack
func azure functionapp publish "$FUNCTION_APP" --python --build remote

## Deploy specific function
func azure functionapp publish "$FUNCTION_APP" --python

## List function app extensions
az functionapp deployment slot list \
  --name "$FUNCTION_APP" \
  --resource-group "$RESOURCE_GROUP"

# ==========================================
# MONITORING & ALERTING
# ==========================================

## View function invocation metrics
az monitor metrics list \
  --resource "/subscriptions/{sub}/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.Web/sites/$FUNCTION_APP" \
  --metric "FunctionExecutionCount"

## Create alert for function failures
az monitor metrics alert create \
  --name "function-failures" \
  --resource-group "$RESOURCE_GROUP" \
  --scopes "/subscriptions/{sub}/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.Web/sites/$FUNCTION_APP" \
  --condition "avg FunctionExecutionCount < 5 over 5m"
